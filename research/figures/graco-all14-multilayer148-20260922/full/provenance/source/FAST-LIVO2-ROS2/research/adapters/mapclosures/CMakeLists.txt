cmake_minimum_required(VERSION 3.16)
project(s3e_mapclosures LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(pybind11 CONFIG REQUIRED)
if(NOT MAPCLOSURES_SOURCE OR NOT HBST_SOURCE OR NOT SOPHUS_SOURCE)
  message(FATAL_ERROR "Provide the three pinned source directories using setup_mapclosures.py")
endif()
# Adapt only a build copy; preserve the exact upstream checkout for provenance.
file(COPY "${MAPCLOSURES_SOURCE}/cpp/" DESTINATION "${CMAKE_CURRENT_BINARY_DIR}/upstream")
set(ransac "${CMAKE_CURRENT_BINARY_DIR}/upstream/map_closures/AlignRansac2D.cpp")
file(READ "${ransac}" code)
string(REPLACE "int iter = 0;" "std::mt19937 generator(0);\n    int iter = 0;" code "${code}")
string(REPLACE "std::mt19937{std::random_device{}()}" "generator" code "${code}")
file(WRITE "${ransac}" "${code}")
set(density "${CMAKE_CURRENT_BINARY_DIR}/upstream/map_closures/DensityMap.cpp")
file(READ "${density}" code)
string(REPLACE "(count - min_points) / (max_points - min_points)"
  "max_points > min_points ? (count - min_points) / (max_points - min_points) : 0.0" code "${code}")
file(WRITE "${density}" "${code}")
set(FETCHCONTENT_SOURCE_DIR_HBST "${HBST_SOURCE}" CACHE PATH "" FORCE)
set(FETCHCONTENT_SOURCE_DIR_SOPHUS "${SOPHUS_SOURCE}" CACHE PATH "" FORCE)
add_subdirectory("${CMAKE_CURRENT_BINARY_DIR}/upstream" upstream-build)
option(S3E_INSPECTION_ONLY "Build a separate inspection extension; preserve the detector binary" OFF)
if(S3E_INSPECTION_ONLY)
  set(module s3e_mapclosures_inspection)
  # Expose the winning RANSAC membership without altering its computations.
  file(READ "${ransac}" code)
  string(REPLACE "PointPair::PointPair(const Eigen::Vector2d &r, const Eigen::Vector2d &q) : ref(r), query(q) {}" "" code "${code}")
  string(REPLACE "RansacAlignment2D(\n    const std::vector<PointPair> &keypoint_pairs)"
    "InspectRansacAlignment2D(\n    const std::vector<PointPair> &keypoint_pairs, std::vector<int> &membership)" code "${code}")
  string(REPLACE "return {T, num_inliers};" "membership = optimal_inlier_indices;\n    return {T, num_inliers};" code "${code}")
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/InspectRansac2D.cpp" "${code}")
  pybind11_add_module(${module} inspect.cpp "${CMAKE_CURRENT_BINARY_DIR}/InspectRansac2D.cpp")
  target_include_directories(${module} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/upstream/map_closures")
else()
  set(module s3e_mapclosures_native)
  pybind11_add_module(${module} adapter.cpp)
endif()
target_compile_definitions(${module} PRIVATE SRRG_HBST_HAS_EIGEN SRRG_HBST_HAS_OPENCV SRRG_MERGE_DESCRIPTORS)
target_include_directories(${module} PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(${module} PRIVATE map_closures)
install(TARGETS ${module} LIBRARY DESTINATION "${PYTHON_MODULE_DIR}")
